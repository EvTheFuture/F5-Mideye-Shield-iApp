            implementation {

# =============================================================================
# Mideye Shield iApp implementation
#
# This block runs every time the iApp is deployed or modified. It:
#   1. Ensures the WHITELIST and BLACKLIST data groups exist (creating them
#      with sensible defaults on first run, leaving them untouched on upgrade
#      so that administrator edits are preserved).
#   2. Generates a salt for username hashing if the administrator left the
#      field blank.
#   3. Builds the three Mideye Shield iRules by taking the iRule bodies that
#      are embedded as TCL "set" assignments below, then substituting every
#      __section__field__ placeholder with the corresponding $::section__field
#      iApp variable, and finally creating or updating each iRule via tmsh.
# =============================================================================

# ---------------------------------------------------------------------------
# Data Groups
#
# MARK-AND-SWEEP SAFETY (this is the important part):
# On reconfigure, the iApp framework re-runs this implementation and then
# deletes (sweeps) any iApp-owned object that was NOT touched (i.e. not
# created/modified) during the run. F5 docs state that untouched objects are
# deleted in the sweep, and that create commands are automatically changed to
# modify on re-entry.
#
# An earlier version guarded BOTH the create and the records with a single
# "if missing" check. That meant on reconfigure the create was skipped, the
# data group became untouched, and mark-and-sweep DELETED it - which then
# broke the iRule with "Unable to find value_list (MIDEYE_SHIELD_BLACKLIST)".
#
# The fix: ALWAYS issue a create for each data group every run (so it is
# always touched and never swept), but only include the seed RECORDS on the
# very first deployment. On reconfigure the create is converted to a modify
# with no records clause, which touches the object while leaving its existing
# records - including any administrator edits - completely intact.
#
# BRACE HANDLING:
# Each create is passed to tmsh::create as a SINGLE brace-grouped argument:
#   tmsh::create { ltm data-group internal ... records add { ... } }
# Wrapping the whole command in braces hands tmsh the literal text with all
# inner braces preserved, so tmsh parses it with its own CLI-style parser
# instead of letting the TCL argument lexer strip the record braces (which
# produced the old "records: required brace is missing" error). These
# commands contain no runtime variables, so a pure brace literal is correct
# here. Backslash-escaped braces are only needed when variables must be
# interpolated into the command, as in the HSSR virtual server command.
# ---------------------------------------------------------------------------

# Helper: return 1 if the given config object is missing, 0 if it exists.
# Handles both get_config failure modes (error thrown OR empty result).
# Note: NO leading slash before "ltm" in the module path (the leading slash
# belongs only on the /Common/<name> object path).
proc _MIDEYE_SHIELD_dg_missing { dg_path } {
    if { [catch { set _cfg [tmsh::get_config ltm data-group internal $dg_path] }] } {
        # get_config threw - object does not exist.
        return 1
    }

    if { [string trim $_cfg] eq "" } {
        # get_config returned nothing - object does not exist.
        return 1
    }

    return 0
}

# Capture each group's pre-run existence BEFORE we touch anything, so we know
# whether to seed default records (first deploy) or just touch the object
# (reconfigure).
set _MIDEYE_SHIELD_wl_missing [_MIDEYE_SHIELD_dg_missing /Common/MIDEYE_SHIELD_WHITELIST]
set _MIDEYE_SHIELD_bl_missing [_MIDEYE_SHIELD_dg_missing /Common/MIDEYE_SHIELD_BLACKLIST]

# WHITELIST: always create (touch), seed RFC1918 defaults only on first deploy.
if { $_MIDEYE_SHIELD_wl_missing } {
    # First-time deployment - create with the default RFC1918 seed records.
    tmsh::create { ltm data-group internal /Common/MIDEYE_SHIELD_WHITELIST type ip records add { 10.0.0.0/8 { } 172.16.0.0/20 { } 192.168.0.0/16 { } } }
} else {
    # Reconfigure - touch the object (create becomes modify, no records
    # clause) so mark-and-sweep does not delete it, preserving admin edits.
    tmsh::create { ltm data-group internal /Common/MIDEYE_SHIELD_WHITELIST type ip }
}

# BLACKLIST: always create (touch). Starts empty; never seed records.
if { $_MIDEYE_SHIELD_bl_missing } {
    # First-time deployment - create empty blacklist.
    tmsh::create { ltm data-group internal /Common/MIDEYE_SHIELD_BLACKLIST type ip }
} else {
    # Reconfigure - touch the object so mark-and-sweep does not delete it,
    # preserving any blacklist entries the administrator has added.
    tmsh::create { ltm data-group internal /Common/MIDEYE_SHIELD_BLACKLIST type ip }
}

# ---------------------------------------------------------------------------
# Username salt
#
# Generate a random salt if the administrator did not supply one. The salt
# is read back from the running config on re-deploy so the value persists.
# ---------------------------------------------------------------------------
if { $::hash__username_salt eq "" } {
    # Pull existing salt from the current iRule (if any) to avoid regenerating
    # on every re-deploy, which would invalidate all previously hashed users.
    set _existing_salt ""

    catch {
        # Mark-and-sweep converts tmsh::create to tmsh::modify on re-deploy,
        # so the rule lives in the iApp folder. tmsh::get_config will find
        # it by short name when run inside the iApp implementation context.
        set _existing_rule [tmsh::get_config /ltm rule MIDEYE_SHIELD_COMMON]
        regexp {static::MIDEYE_SHIELD_username_salt\s+"([^"]*)"} $_existing_rule -> _existing_salt    ;# Trailing " to keep an even number of unescaped double quotes on this line, so tmsh's outer parser does not lose track of string state when reading the iApp implementation block.
    }

    if { $_existing_salt ne "" } {
        set ::hash__username_salt $_existing_salt
    } else {
        # Generate a fresh 32-character random salt
        set _chars "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
        set _salt ""

        for { set _i 0 } { $_i < 32 } { incr _i } {
            append _salt [string index $_chars [expr { int(rand() * [string length $_chars]) }]]
        }

        set ::hash__username_salt $_salt
    }
}

# ---------------------------------------------------------------------------
# Internal substitution setup
#
# Compute the values of internal placeholder substitutions (e.g. partition
# name from tmsh::pwd) before the substitution map is built. Each TCL
# variable defined here is referenced by an entry in _SUBST_MAP below.
# ---------------------------------------------------------------------------
__IAPP_INTERNAL_SETUP__

# ---------------------------------------------------------------------------
# Build placeholder substitution map
#
# Every placeholder in the iRule bodies (e.g. __section__field__ for user
# settings, or __partition__ / __hssr_irule__ for internal substitutions)
# is mapped to its concrete value here. The map is built once and applied
# to all iRule bodies below.
# ---------------------------------------------------------------------------
set _SUBST_MAP [list __IAPP_PLACEHOLDER_MAP_ENTRIES__]

# ---------------------------------------------------------------------------
# iRule bodies
#
# The generator script embeds each iRule's TCL source as a "set <name> { ... }"
# block here. The placeholders inside each body are substituted from the
# _SUBST_MAP above before the iRule is created.
# ---------------------------------------------------------------------------
__IAPP_IRULE_BODIES__

# ---------------------------------------------------------------------------
# Create or update each iRule
#
# tmsh::create is automatically converted to tmsh::modify on iApp re-entry
# by the BIG-IP "mark-and-sweep" framework, so a single call works for both
# first deployment and re-deploys. The framework also handles deletion of
# orphaned objects on iApp delete, so no explicit cleanup is needed.
# ---------------------------------------------------------------------------
__IAPP_IRULE_CREATE_CALLS__

            }
            macro {
            }

__IAPP_PRESENTATION__

__IAPP_TEXT__

        }
