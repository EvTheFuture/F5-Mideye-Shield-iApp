# =============================================================================
# Mideye Shield iApp implementation

# ---------------------------------------------------------------------------
# Data Groups
#
# The WHITELIST and BLACKLIST data groups are
#   (a) auto-created with sensible defaults on first deploy,
#   (b) freely editable by the administrator afterwards (tmsh / GUI), and
#   (c) NEVER modified or deleted by the iApp on reconfigure or upgrade.
#
# The groups are created in the partition the application service is deployed
# into (e.g. /Common), NOT the iApp app-folder (/Common/MIDEYE_SHIELD.app),
# so they are resolvable by the iRules and independent of the iApp folder. The
# partition is derived from tmsh::pwd into the variable "partition" just below
# and referenced as /$partition/<name>.
# ---------------------------------------------------------------------------

# Resolve the PARTITION the application service is being deployed into (e.g.
# Common), NOT the iApp app-folder (MIDEYE_SHIELD.app). tmsh::pwd returns the
# app-folder path like /Common/MIDEYE_SHIELD.app; splitting on "/" gives
# {"" "Common" "MIDEYE_SHIELD.app"} so index 1 is the partition name. Fall
# back to Common if anything is unexpected.
set partition [lindex [split [tmsh::pwd] "/"] 1]

if { $partition eq "" } {
    set partition "Common"
}

# Helper: return 1 if the given internal data group is missing, else 0.
# Runs via exec tmsh (outside the iApp transaction) and treats any error or
# empty output as missing.
proc _MIDEYE_SHIELD_dg_missing { dg_path } {
    if { [catch { set _out [exec tmsh list ltm data-group internal $dg_path] } _err] } {
        # tmsh exited non-zero (object not found) - treat as missing.
        return 1
    }

    if { [string trim $_out] eq "" } {
        # No output - treat as missing.
        return 1
    }

    return 0
}

# WHITELIST: create with RFC1918 defaults ONLY if absent. Created by piping
# the command into a separate tmsh -a process, so the group is unowned and
# admin-editable and never swept. Existing groups are left alone.
if { [_MIDEYE_SHIELD_dg_missing /$partition/MIDEYE_SHIELD_WHITELIST] } {
    exec tmsh -a create ltm data-group internal /$partition/MIDEYE_SHIELD_WHITELIST type ip records add \{ 10.0.0.0/8 \{ \} 172.16.0.0/20 \{ \} 192.168.0.0/16 \{ \} \}
}

# BLACKLIST: create empty ONLY if absent, same out-of-transaction method.
if { [_MIDEYE_SHIELD_dg_missing /$partition/MIDEYE_SHIELD_BLACKLIST] } {
    exec tmsh -a create ltm data-group internal /$partition/MIDEYE_SHIELD_BLACKLIST type ip
}

# ---------------------------------------------------------------------------
# Username salt
#
# Generate a random salt if the administrator did not supply one. The salt
# is read back from the running config on re-deploy so the value persists.
# ---------------------------------------------------------------------------
if { $::hash__username_salt eq "" } {
    # Pull existing salt from the current iRule (if any) to avoid regenerating
    # on every re-deploy, which would invalidate all previously hashed users.
    set _existing_salt ""

    catch {
        # Mark-and-sweep converts tmsh::create to tmsh::modify on re-deploy,
        # so the rule lives in the iApp folder. tmsh::get_config will find
        # it by short name when run inside the iApp implementation context.
        set _existing_rule [tmsh::get_config /ltm rule MIDEYE_SHIELD_COMMON]
        regexp {static::MIDEYE_SHIELD_username_salt\s+"([^"]*)"} $_existing_rule -> _existing_salt    ;# Trailing " to keep an even number of unescaped double quotes on this line, so tmsh's outer parser does not lose track of string state when reading the iApp implementation block.
    }

    if { $_existing_salt ne "" } {
        set ::hash__username_salt $_existing_salt
    } else {
        # Generate a fresh 32-character random salt
        set _chars "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
        set _salt ""

        for { set _i 0 } { $_i < 32 } { incr _i } {
            append _salt [string index $_chars [expr { int(rand() * [string length $_chars]) }]]
        }

        set ::hash__username_salt $_salt
    }
}

# Internal substitution setup
#
# Compute the values of internal placeholder substitutions (e.g. partition
# name from tmsh::pwd) before the substitution map is built. Each TCL
# variable defined here is referenced by an entry in _SUBST_MAP below.
# ---------------------------------------------------------------------------
__IAPP_INTERNAL_SETUP__

# ---------------------------------------------------------------------------
# Build placeholder substitution map
#
# Every placeholder in the iRule bodies (e.g. __section__field__ for user
# settings, or __partition__ / __hssr_irule__ for internal substitutions)
# is mapped to its concrete value here. The map is built once and applied
# to all iRule bodies below.
# ---------------------------------------------------------------------------
set _SUBST_MAP [list __IAPP_PLACEHOLDER_MAP_ENTRIES__]

# ---------------------------------------------------------------------------
# iRule bodies
#
# The generator script embeds each iRule's TCL source as a "set <name> { ... }"
# block here. The placeholders inside each body are substituted from the
# _SUBST_MAP above before the iRule is created.
# ---------------------------------------------------------------------------
__IAPP_IRULE_BODIES__

# Create or update each iRule
#
# tmsh::create is automatically converted to tmsh::modify on iApp re-entry
# by the BIG-IP "mark-and-sweep" framework, so a single call works for both
# first deployment and re-deploys. The framework also handles deletion of
# orphaned objects on iApp delete, so no explicit cleanup is needed.
# ---------------------------------------------------------------------------
__IAPP_IRULE_CREATE_CALLS__
